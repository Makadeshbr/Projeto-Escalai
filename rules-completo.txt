# Regras RLS Completas para EscalaiApp
# Aplique estas regras no painel da Aether em Database > Rules

rules_version = '2';

service cloud.database {
  
  // ============================================
  // 1. driver_status - Controle de Motoristas
  // ============================================
  match /driver_status/{document=**} {
    allow read: if auth != null;
    // Admin pode tudo, motorista pode criar/atualizar próprio registro
    allow write: if auth != null && (auth.role == 'admin' || auth.uid == resource.id || auth.uid == request.resource.id);
  }
  
  // ============================================
  // 2. driver_availability - Respostas dos Motoristas
  // ============================================
  match /driver_availability/{document=**} {
    allow read: if auth != null;
    // Admin pode tudo, motorista pode criar/atualizar próprio registro
    allow write: if auth != null && (auth.role == 'admin' || auth.uid == request.resource.data.driverId || auth.uid == resource.data.driverId);
  }
  
  // ============================================
  // 3. assignments - Despachos/Rotas
  // ============================================
  match /assignments/{document=**} {
    allow read: if auth != null;
    // Admin pode tudo, motorista pode ler seus próprios assignments
    allow write: if auth != null && (auth.role == 'admin' || auth.uid == request.resource.data.driverId || auth.uid == resource.data.driverId);
  }
  
  // ============================================
  // 4. availability_windows - Janelas de Disponibilidade
  // ============================================
  match /availability_windows/{document=**} {
    allow read: if auth != null;
    // Apenas admin pode criar/modificar janelas
    allow write: if auth != null && auth.role == 'admin';
  }
  
  // ============================================
  // 5. cities - Cidades
  // ============================================
  match /cities/{document=**} {
    allow read: if auth != null;
    // Apenas admin pode modificar cidades
    allow write: if auth != null && auth.role == 'admin';
  }
  
}
