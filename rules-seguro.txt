rules_version = '2';

service cloud.database {
  
  // Cidades - só admin
  match /cities/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null && auth.role == 'admin';
  }

  // Janelas - só admin
  match /availability_windows/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null && auth.role == 'admin';
  }

  // Status Motoristas - qualquer usuário logado pode criar/editar (permissivo)
  match /driver_status/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null;
  }

  // Despachos - qualquer usuário logado (mais permissivo)
  match /assignments/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null;
  }

  // Disponibilidade Motoristas - qualquer usuário logado
  match /driver_availability/{document=**} {
    allow read, write: if auth != null;
  }

  // Retrocompatibilidade
  match /routes/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null && auth.role == 'admin';
  }

  match /availabilities/{document=**} {
    allow read, write: if auth != null;
  }
  
}
