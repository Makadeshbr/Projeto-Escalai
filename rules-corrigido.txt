rules_version = '2';

service cloud.database {
  
  // [PT-BR] 1. Cidades Base
  match /cities/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null && auth.role == 'admin';
  }

  // [PT-BR] 2. Janelas de Disponibilidade
  match /availability_windows/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null && auth.role == 'admin';
  }

  // [PT-BR] 3. Status/Controle de Motoristas
  // driver_status usa o campo 'user_id' (não driverId)
  match /driver_status/{document=**} {
    allow read: if auth != null;
    // Motorista pode criar se user_id for igual ao seu uid
    allow create: if auth != null && request.resource.data.user_id == auth.uid;
    // Admin ou próprio usuário pode editar
    allow update, delete: if auth != null && (auth.role == 'admin' || resource.data.user_id == auth.uid);
  }

  // [PT-BR] 4. Despachos / Atribuições de Rota
  // assignments usa o campo 'driverId'
  match /assignments/{document=**} {
    allow read: if auth != null;
    // Admin pode tudo; Motorista pode criar se driverId for igual ao seu uid
    allow create: if auth != null && (auth.role == 'admin' || request.resource.data.driverId == auth.uid);
    // Admin ou próprio motorista pode editar
    allow update, delete: if auth != null && (auth.role == 'admin' || resource.data.driverId == auth.uid);
  }

  // [PT-BR] 5. Escalas dos Motoristas
  // driver_availability usa o campo 'driverId'
  match /driver_availability/{document=**} {
    allow read: if auth != null;
    // Motorista pode criar se driverId for igual ao seu uid
    allow create: if auth != null && request.resource.data.driverId == auth.uid;
    // Admin ou próprio motorista pode editar
    allow update, delete: if auth != null && (auth.role == 'admin' || resource.data.driverId == auth.uid);
  }

  // [PT-BR] 6. Retrocompatibilidade
  match /routes/{document=**} {
    allow read: if auth != null;
    allow write: if auth != null && auth.role == 'admin';
  }

  match /availabilities/{document=**} {
    allow read: if auth != null;
    allow create: if auth != null && request.resource.data.driverId == auth.uid;
    allow update, delete: if auth != null && (auth.role == 'admin' || resource.data.driverId == auth.uid);
  }
  
}
